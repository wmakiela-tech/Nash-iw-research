Code:
```python
import numpy as np
for L in [1,2,4,8]:
    idx=np.arange(-L,L+1); out=np.arange(-L-1,L+2)
    sq=np.zeros((len(idx),len(idx)))
    rect=np.zeros((len(out),len(idx)))
    for j,k in enumerate(idx):
        if k+1 in idx: sq[np.where(idx==k+1)[0][0],j]=1
        rect[np.where(out==k+1)[0][0],j]=1
    assert np.allclose(np.linalg.eigvals(sq),0,atol=1e-12)
    assert np.isclose(np.linalg.svd(rect,compute_uv=False).min(),1)
```
